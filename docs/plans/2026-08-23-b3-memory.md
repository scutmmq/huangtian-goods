# B3 — AI 助手长期记忆系统实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为荒天享物商城 AI 助手实现用户级长期记忆(身份档案 + 偏好画像),事件驱动 + MySQL/Redis 双层 + 完整 GDPR 防御,业务流零退化。

**Architecture:** 14 个新类,职责清晰;MyBatis-Plus + Redisson + Guava RateLimiter + Spring `@Async` + Micrometer;事件驱动写、cache-aside 读;PromptSanitizer 3 层防御注入;`compute_seq` 抗 TOCTOU;Redisson raw-lock + Watchdog 防 cron 锁卡死;Async audit purge + DLQ 满足 GDPR Art 17。

**Tech Stack:** Spring Boot 3.5.5 / Java 17 / MyBatis-Plus 3.5 / MySQL 8 / Redisson / Guava RateLimiter / Micrometer + Prometheus

- **命名约定**:包路径见 v0.3 spec §3.0;camelCase Java / snake_case DB;表名 `ai_*` 前缀

- **业务流零退化**:70/70 现有单测必须通过,所有 C0-C13 Eval YAML 仍 PASS

```java
package com.scutmmq.ai.config;

import jakarta.annotation.PostConstruct;
import lombok.Data;
import org.springframework.boot.context.properties.ConfigurationProperties;
import org.springframework.stereotype.Component;

@Data
@Component
@ConfigurationProperties(prefix = "ai.memory")
public class AiMemoryProperties {
    private String cacheHmacSecrets = "v1:dev-only-fallback-please-set-via-env-in-production-min-32-chars";
    private String activeSecretVersion = "v1";
    private int coalesceTtlSeconds = 60;
    private String recomputeCron = "0 0 3 * * ?";
    private String partitionDropCron = "0 0 2 1 * ?";
    private int promptTokenCap = 600;
    private int recomputeBatchSize = 1000;
    private int recomputeMaxFailCount = 3;
    private int resetRetentionDays = 180;
    private int auditPartitionRetentionDays = 90;
    private int auditPurgeRateRowsPerSec = 100;
    private int localRateLimitPerUser = 1;
    private int localRateLimitBurst = 200;

    @PostConstruct
    public void validate() {
        if (cacheHmacSecrets == null || cacheHmacSecrets.length() < 32) {
            throw new IllegalStateException("ai.memory.cache-hmac-secrets must be >= 32 chars");
        }
        if (activeSecretVersion == null || !cacheHmacSecrets.contains(activeSecretVersion + ":")) {
            throw new IllegalStateException("ai.memory.active-secret-version '" + activeSecretVersion + "' not present in cache-hmac-secrets");
        }
    }

    /** 解析 active secret(去掉 v{N}: 前缀) */
    public String getActiveSecret() {
        for (String entry : cacheHmacSecrets.split(",")) {
            String[] kv = entry.trim().split(":", 2);
            if (kv.length == 2 && kv[0].trim().equals(activeSecretVersion)) {
                return kv[1].trim();
            }
        }
        throw new IllegalStateException("active secret not found: " + activeSecretVersion);
    }
}
```

```java
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Component;

import javax.crypto.Mac;
import javax.crypto.spec.SecretKeySpec;
import java.nio.charset.StandardCharsets;
import java.time.Instant;
import java.util.HexFormat;
import java.util.Optional;

@Slf4j
@Component
@RequiredArgsConstructor
public class UserMemoryCache {

    private final StringRedisTemplate redis;
    private final AiMemoryProperties props;
    private final ObjectMapper mapper;

    public record CacheSnapshot(String identityJson, String preferenceJson, long computeSeq, Instant computedAt) {}

    /** HMAC-SHA256(activeSecret, userId.toString()) 截 16 hex */
    public String hmacKey(Long userId) {
        try {
            String secret = props.getActiveSecret();
            Mac mac = Mac.getInstance("HmacSHA256");
            mac.init(new SecretKeySpec(secret.getBytes(StandardCharsets.UTF_8), "HmacSHA256"));
            byte[] hash = mac.doFinal(userId.toString().getBytes(StandardCharsets.UTF_8));
            return RedisConstants.MEMORY_CACHE_KEY_PREFIX
                + "v" + props.getActiveSecretVersion().substring(1) + ":"
                + HexFormat.of().formatHex(hash).substring(0, 16);
        } catch (Exception e) {
            throw new IllegalStateException("HMAC failure", e);
        }
    }

    public Optional<CacheSnapshot> get(Long userId) {
        try {
            String raw = redis.opsForValue().get(hmacKey(userId));
            return raw == null ? Optional.empty() : Optional.of(deserialize(raw));
        } catch (Exception e) {
            log.warn("[AI][MEMORY] redis get failed userId={} reason={}", userId, e.getMessage());
            return Optional.empty();  // 降级
        }
    }

    public void setIfAbsentNewer(Long userId, CacheSnapshot snapshot) {
        try {
            String key = hmacKey(userId);
            String json = serialize(snapshot);
            // Redis SETNX + 检查 compute_seq(TOCTOU 防护)
            // 用 Lua 脚本保证原子性
            String script = "local cur = redis.call('GET', KEYS[1]) " +
                            "if cur == false or tonumber(string.match(cur, '\"computeSeq\":(%d+)')) < tonumber(ARGV[1]) " +
                            "then return redis.call('SET', KEYS[1], ARGV[2], 'EX', ARGV[3]) else return false end";
            redis.execute((connection) -> connection.scriptingCommands().eval(
                script.getBytes(), org.springframework.data.redis.connection.ReturnType.BOOLEAN, 1,
                key.getBytes(), json.getBytes(),
                String.valueOf(snapshot.computeSeq()).getBytes(),
                String.valueOf(props.getPromptTokenCap() * 6).getBytes()  // TTL = token cap * 6 s,粗略
            ), true);
        } catch (Exception e) {
            log.warn("[AI][MEMORY] cache set failed userId={} reason={}", userId, e.getMessage());
        }
    }

    public void invalidate(Long userId) {
        try { redis.delete(hmacKey(userId)); } catch (Exception ignore) {}
    }

    public String serialize(CacheSnapshot s) {
        try { return mapper.writeValueAsString(s); }
        catch (JsonProcessingException e) { throw new IllegalStateException(e); }
    }

    public CacheSnapshot deserialize(String json) {
        try { return mapper.readValue(json, CacheSnapshot.class); }
        catch (JsonProcessingException e) { throw new IllegalStateException(e); }
    }
}
```

```java
package com.scutmmq.ai.security;

import org.apache.commons.text.StringEscapeUtils;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.regex.Pattern;

@Component
public class PromptSanitizer {

    public enum FieldType { CATEGORY_NAME, MERCHANT_NAME, FREE_TEXT }

    private static final List<Pattern> DENY_LIST = List.of(
        Pattern.compile("<\\|.*?\\|>"),
        Pattern.compile("(?i)ignore\\s+(previous|all|above)"),
        Pattern.compile("(?i)system\\s*:\\s*"),
        Pattern.compile("(?i)assistant\\s*:\\s*"),
        Pattern.compile("(?i)you\\s+are\\s+now"),
        Pattern.compile("(?i)disregard\\s+(all|previous)")
    );

    private static final Pattern SAFE_NAME = Pattern.compile(
        "^[一-龥a-zA-Z0-9\\s\\-_()&（）【】]{1,32}$");

    public String sanitize(String raw, FieldType type) {
        if (raw == null || raw.isEmpty()) return "";
        for (Pattern p : DENY_LIST) {
            if (p.matcher(raw).find()) {
                throw new PromptInjectionException("Deny-list match: " + p.pattern() + " input=" + raw);
            }
        }
        if ((type == FieldType.CATEGORY_NAME || type == FieldType.MERCHANT_NAME)
            && !SAFE_NAME.matcher(raw).matches()) {
            return "[FILTERED]";
        }
        return StringEscapeUtils.escapeJson(raw);
    }
}
```

```java
@Slf4j
@Service
@RequiredArgsConstructor
public class UserMemoryService {

    private final StringRedisTemplate redis;
    private final UserMemoryMapper mapper;
    private final UserMemoryBuilder builder;
    private final UserMemoryCache cache;
    private final PromptSanitizer sanitizer;
    private final AuditService auditService;
    private final AiMemoryProperties props;
    @Qualifier("memoryAsyncExecutor")
    private final Executor asyncExecutor;

    private final ConcurrentHashMap<Long, RateLimiter> localLimiters = new ConcurrentHashMap<>();

    public boolean scheduleRecompute(Long userId, TriggerReason reason) {
        String key = RedisConstants.MEMORY_COALESCE_KEY_PREFIX + userId;
        try {
            Boolean acquired = redis.opsForValue().setIfAbsent(
                key, Instant.now().toString(),
                Duration.ofSeconds(props.getCoalesceTtlSeconds()));
            if (Boolean.TRUE.equals(acquired)) {
                asyncExecutor.execute(() -> recomputeFor(userId, reason));
                return true;
            }
            return false;
        } catch (Exception e) {
            return scheduleWithLocalRateLimit(userId, reason);
        }
    }

    private boolean scheduleWithLocalRateLimit(Long userId, TriggerReason reason) {
        RateLimiter limiter = localLimiters.computeIfAbsent(userId,
            uid -> RateLimiter.create(props.getLocalRateLimitPerUser()));
        if (limiter.tryAcquire(0, TimeUnit.MILLISECONDS)) {
            auditService.logDegraded(userId, "DEGRADED_RATE_LIMITED");
            asyncExecutor.execute(() -> recomputeFor(userId, reason));
            return true;
        }
        auditService.logDegraded(userId, "DEGRADED_RATE_LIMITED_DROP");
        return false;
    }

    public UserMemorySnapshot recomputeFor(Long userId, TriggerReason reason) {
        try {
            // 1. 查 user memory(获得 version for @Version)
            UserMemoryEntity entity = mapper.selectById(userId);
            int currentVersion = entity == null ? 1 : entity.getVersion();

            // 2. computeIdentity + updateIdentity(MP interceptor 乐观锁)
            UserMemorySnapshot identity = builder.computeIdentity(userId);
            if (entity == null) {
                entity = new UserMemoryEntity();
                entity.setUserId(userId);
                entity.setIdentityJson(identity.identityJson());
                entity.setPreferenceJson("{}");
                entity.setComputedAt(LocalDateTime.now());
                mapper.insert(entity);
            } else {
                mapper.updateIdentity(userId, identity.identityJson(), currentVersion);
            }
            // 3. computePreference + updatePreference
            UserMemorySnapshot pref = builder.computePreference(userId);
            mapper.updatePreference(userId, pref.preferenceJson(), currentVersion + 1);
            // 4. bumpComputeSeq
            mapper.bumpComputeSeq(userId, currentVersion + 1);
            // 5. invalidate cache
            cache.invalidate(userId);
            // 6. audit
            auditService.logCompute(userId, reason);
        } catch (OptimisticLockingFailureException e) {
            // retry 1 次
            log.warn("[AI][MEMORY] optimistic lock fail userId={}, retrying", userId);
            // ... retry
        } catch (DataIntegrityViolationException e) {
            if (e.getMessage().contains("chk_preference_size") || e.getMessage().contains("chk_identity_size")) {
                auditService.logJsonOverflow(userId, e.getMessage().contains("identity") ? "identity" : "preference");
                mapper.updateIdentity(userId, "{}", currentVersion);
                mapper.updatePreference(userId, "{}", currentVersion + 1);
            }
        } catch (Exception e) {
            // fail_count++
            mapper.incrementFailCount(userId);
            UserMemoryEntity latest = mapper.selectById(userId);
            if (latest != null && latest.getFailCount() >= props.getRecomputeMaxFailCount()) {
                mapper.markDisabled(userId);
            }
            auditService.logRecomputeFail(userId, e);
        }
        return null;
    }

    public String renderMemorySection(Long userId) {
        Optional<CacheSnapshot> cached = cache.get(userId);
        UserMemorySnapshot snap;
        long cachedSeq = 0;
        if (cached.isPresent()) {
            cachedSeq = cached.get().computeSeq();
            // 验证 seq 仍然有效
            Long dbSeq = mapper.getComputeSeq(userId);
            if (dbSeq != null && dbSeq <= cachedSeq) {
                snap = new UserMemorySnapshot(cached.get().identityJson(), cached.get().preferenceJson());
            } else {
                // seq 过期,回源
```

```java
    @Async("memoryAsyncExecutor")
    public void purgeAuditAsync(Long userId) {
        RateLimiter limiter = RateLimiter.create(props.getAuditPurgeRateRowsPerSec());
        int batchSize = 1000;
        int retry = 0;
        while (retry < 3) {
            try {
                Integer total = jdbc.queryForObject(
                    "SELECT COUNT(*) FROM ai_user_memory_audit WHERE user_id=? AND action IN ('COMPUTE','OVERFLOW_DROP')",
                    Integer.class, userId);
                for (int offset = 0; offset < total; offset += batchSize) {
                    limiter.acquire(batchSize);
                    jdbc.update(
                        "UPDATE ai_user_memory_audit SET user_id = 0, fields_changed = NULL WHERE user_id=? AND action IN ('COMPUTE','OVERFLOW_DROP') LIMIT " + batchSize,
                        userId);
                }
                return;
            } catch (Exception e) {
                retry++;
                if (retry >= 3) {
                    jdbc.update("INSERT INTO ai_user_memory_audit_purge_dlq(user_id, error_msg, retry_count) VALUES (?,?,?)",
                        userId, e.getMessage(), retry);
                }
            }
        }
    }
}
```
